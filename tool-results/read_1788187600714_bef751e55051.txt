    15→/** 1 px CSS = 0,264583 mm (96 DPI) — dasar perhitungan skala cetak A4. */
    16→const MM_PER_PX = 25.4 / 96;
    17→/** Meter per piksel pada zoom 0 Web Mercator (di ekuator). */
    18→const MPP_Z0 = 156543.03392;
    19→const ZOOM_MAKS = 25; // di atas ZOOM_TILE_ASLI citra di-upscale digital
    20→const ZOOM_TILE_ASLI = 19;
    21→
    22→const derajatKeRadian = (d: number) => (d * Math.PI) / 180;
    23→
    24→/** Zoom pecahan agar 1 px kertas = MM_PER_PX × S mm di lapangan (skala 1:S). */
    25→function zoomUntukSkala(s: number, lat: number, lebarPx: number): number {
    26→  const mpp = (MM_PER_PX * s) / 1000; // meter lapangan per piksel kertas
    27→  return Math.log2((MPP_Z0 * Math.cos(derajatKeRadian(lat))) / mpp);
    28→}
    29→
    30→/** Skala 1:n aktual dari zoom saat ini pada latitude pusat peta. */
    31→function skalaDariZoom(z: number, lat: number): number {
    32→  const mpp = (MPP_Z0 * Math.cos(derajatKeRadian(lat))) / Math.pow(2, z);
    33→  return Math.round((mpp * 1000) / MM_PER_PX);
    34→}
    35→
    36→/** Terima "1:50", "1/50", "50", "1 : 350" → angka pembagi skala. */
    37→function parseSkala(teks: string): number | null {
    38→  const b = teks.trim().replace(/,/g, ".").replace(/\s+/g, "");
    39→  if (!b) return null;
    40→  const m = b.match(/^1[:/x](\d+(?:\.\d+)?)$/i);
    41→  if (m) return parseFloat(m[1]);
    42→  const n = parseFloat(b);
    43→  return Number.isFinite(n) && n >= 1 ? n : null;
    44→}
    45→
    46→const formatAngka = (n: number) => Math.round(n).toLocaleString("id-ID");
    47→
    48→const tanggalKini = () => {
    49→  const d = new Date();
    50→  const p = (x: number) => String(x).padStart(2, "0");
    51→  return `${p(d.getDate())}/${p(d.getMonth() + 1)}/${d.getFullYear()}`;
    52→};

    80→  const mapRef = useRef<L.Map | null>(null);
    81→  const layerRef = useRef<L.LayerGroup | null>(null);

    84→  const pernahKetikRef = useRef(false);
    85→  const terapkanRef = useRef<(s: number) => void>(() => {});

   101→      const c = map.getCenter();
   102→      setSkalaKini(skalaDariZoom(map.getZoom(), c.lat));
   103→      setCitraUpscale(map.getZoom() > ZOOM_TILE_ASLI + 0.01);
   104→    };

   106→    map.on("moveend", perbaruiSkala);
   107→    mapRef.current = map;
   108→    layerRef.current = L.layerGroup().addTo(map);
   109→    (window as unknown as Record<string, unknown>).__layoutMap = map;
   110→    setTimeout(() => {
   111→      map.invalidateSize();
   112→      perbaruiSkala();

   114→
   115→    return () => {
   116→      map.remove();
   117→      mapRef.current = null;
   118→      layerRef.current = null;
   119→      delete (window as unknown as Record<string, unknown>).__layoutMap;
   120→    };

   122→
   123→  // ---------- tile basemap layout ----------
   124→  useEffect(() => {
   125→    const map = mapRef.current;
   126→    if (view !== "layout" || !mapDiv || !map) return;
   127→    map.eachLayer((l) => {
   128→      if (l instanceof L.TileLayer) map.removeLayer(l);
   129→    });

   131→      basemapLayout === "sat"
   132→        ? L.tileLayer("https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}", { maxNativeZoom: ZOOM_TILE_ASLI, maxZoom: ZOOM_MAKS })
   133→        : L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", { maxNativeZoom: ZOOM_TILE_ASLI, maxZoom: ZOOM_MAKS });

   135→    tile.bringToBack();

   137→
   138→  // ---------- render lapisan data ----------
   139→  useEffect(() => {
   140→    const map = mapRef.current;
   141→    const layer = layerRef.current;
   142→    if (view !== "layout" || !mapDiv || !map || !layer) return;
   143→    layer.clearLayers();
   144→
   145→    if (lapisan.kontur) {
   146→      let eMin = Infinity;
   147→      let eMax = -Infinity;
   148→      for (const c of contours)
   149→        for (const p of c.paths) {
   150→          if (p.elev < eMin) eMin = p.elev;
   151→          if (p.elev > eMax) eMax = p.elev;
   152→        }
   153→      const rentang = Math.max(eMax - eMin, 1e-6);
   154→      for (const c of contours) {
   155→        if (!c.visible) continue;
   156→        for (const path of c.paths) {
   157→          L.polyline(
   158→            path.coords.map((x) => [x.lat, x.lng] as [number, number]),
   159→            { color: warnaElevasi((path.elev - eMin) / rentang), weight: 1.5, opacity: 0.9 }

   161→        }
   162→      }
   163→    }

   165→      for (const sh of shapes) {
   166→        if (!sh.visible) continue;

   168→        if (sh.kind === "closed" && latlngs.length >= 3) {
   169→          L.polygon(latlngs, { color: sh.color, weight: 2, fillOpacity: 0.15 }).addTo(layer);
   170→        } else if (latlngs.length >= 2) {
   171→          L.polyline(latlngs, { color: sh.color, weight: 2.5 }).addTo(layer);
   172→        }
   173→      }
   174→    }

   176→      const renderer = L.canvas({ padding: 0.3 });

   183→          fillOpacity: 0.9,
   184→          weight: 1,
   185→        })
   186→          .bindTooltip(p.title, {
   187→            permanent: points.length <= 60,
   188→            direction: "top",
   189→            className: "geokita-contour-label",
   190→          })
   191→          .addTo(layer);
   192→      }
   193→    }
   194→    if (lapisan.label) {
   195→      for (const lb of labels) {
   196→        L.marker([lb.lat, lb.lng], {
   197→          icon: L.divIcon({
   198→            className: "",
   199→            html: `<div class="geokita-label">${lb.text.replace(/</g, "&lt;")}</div>`,
   200→          }),
   201→        }).addTo(layer);
   202→      }
   203→    }
   204→
   205→    const semua: [number, number][] = [
   206→      ...points.map((p) => [p.lat, p.lng] as [number, number]),
   207→      ...shapes.flatMap((s) => s.vertices.map((v) => [v.lat, v.lng] as [number, number])),
   208→    ];
   209→    if (semua.length > 0 && modeSkalaRef.current === "auto") map.fitBounds(L.latLngBounds(semua).pad(0.1));
   210→    setTimeout(() => map.invalidateSize(), 80);
   211→  }, [points, shapes, labels, contours, lapisan, view, mapDiv]);
   212→
   213→  // ---------- terapkan skala cetak (1:n) ----------
   214→  const terapkanSkala = (s: number) => {
   215→    const map = mapRef.current;
   216→    if (!map || !mapDiv || mapDiv.clientWidth <= 0) return;
   217→    const c = map.getCenter();
   218→    const zIdeal = zoomUntukSkala(s, c.lat, mapDiv.clientWidth);
   219→    const z = Math.min(Math.max(zIdeal, 0), ZOOM_MAKS);
   220→    setModeSkala("manual");
   221→    modeSkalaRef.current = "manual";
   222→    lastAppliedRef.current = s;
   223→    map.setZoom(z);
   224→    setSkalaKini(skalaDariZoom(z, c.lat));
   225→    if (zIdeal - z > 0.01) {
   226→      toast.warning(`Skala 1:${formatAngka(s)} di luar jangkauan`, {
   227→        description: `Peta dibatasi ke skala ≈ 1:${formatAngka(skalaDariZoom(z, c.lat))} (zoom maksimum).`,
   228→      });
   229→    } else if (z > ZOOM_TILE_ASLI) {
   230→      toast.info(`Skala 1:${formatAngka(s)} diterapkan`, {
   231→        description: "Citra dasar diperbesar melebihi resolusi aslinya (agak buram), seperti resampling di ArcGIS.",
   232→      });
   233→    } else {
   234→      toast.success(`Skala peta: 1:${formatAngka(s)}`);
   235→    }
   236→  };
   237→
   238→  useEffect(() => {
   239→    terapkanRef.current = terapkanSkala;
   240→  });
   241→
   242→  // selesai mengetik skala → otomatis diterapkan (debounce 700 ms)
   243→  useEffect(() => {
   244→    if (!pernahKetikRef.current) return;
   245→    const t = setTimeout(() => {
   246→      const s = parseSkala(skalaInput);
   247→      if (s && s !== lastAppliedRef.current) terapkanRef.current(s);
   248→    }, 700);
   249→    return () => clearTimeout(t);
   250→  }, [skalaInput]);
   251→
   252→  // ganti orientasi → peta sesuaikan ukuran kertas baru (ulang skala/fit)
   253→  useEffect(() => {
   254→    if (view !== "layout" || !mapRef.current) return;
   255→    const t = setTimeout(() => {
   256→      const map = mapRef.current;
   257→      if (!map) return;
   258→      map.invalidateSize();
   259→      if (modeSkalaRef.current === "manual" && lastAppliedRef.current) {
   260→        terapkanRef.current(lastAppliedRef.current);
   261→      } else {
   262→        const st = useGis.getState();
   263→        const semua: [number, number][] = [
   264→          ...st.points.map((p) => [p.lat, p.lng] as [number, number]),
   265→          ...st.shapes.flatMap((x) => x.vertices.map((v) => [v.lat, v.lng] as [number, number])),
   266→        ];
   267→        if (semua.length > 0) map.fitBounds(L.latLngBounds(semua).pad(0.1));
   268→      }

   271→  }, [orientasi, view]);
   272→

   275→    ? `Skala ${skalaKini ? "1/" + formatAngka(skalaKini) : "±"} • tanggal ${tanggalKini()}`
   276→    : subJudul;
   277→
   278→  if (view !== "layout") return null;
   279→
   280→  const ukuran = UKURAN[orientasi];
   281→  const legendaItems = [
   282→    lapisan.titik && points.length > 0
   283→      ? { warna: "#3b82f6", label: `Titik (${points.length.toLocaleString("id-ID")})` }
   284→      : null,
   285→    lapisan.bentuk && shapes.some((s) => s.kind === "closed")
   286→      ? { warna: shapes.find((s) => s.kind === "closed")?.color ?? "#f59e0b", label: "Poligon" }
   287→      : null,
   288→    lapisan.bentuk && shapes.some((s) => s.kind === "open")
   289→      ? { warna: shapes.find((s) => s.kind === "open")?.color ?? "#10b981", label: "Garis" }
   290→      : null,
   291→    lapisan.kontur && contours.length > 0 ? { warna: warnaElevasi(0.5), label: "Kontur" } : null,
   292→  ].filter(Boolean) as { warna: string; label: string }[];

   297→  };
   298→
   299→  const pilihPreset = (s: number) => {
   300→    pernahKetikRef.current = true;
   301→    setSkalaInput(`1:${s}`);
   302→    terapkanSkala(s);
   303→  };
   304→
   305→  const pasOtomatis = () => {
   306→    setModeSkala("auto");
   307→    modeSkalaRef.current = "auto";
   308→    lastAppliedRef.current = null;
   309→    const map = mapRef.current;
   310→    if (!map) return;
   311→    const semua: [number, number][] = [
   312→      ...points.map((p) => [p.lat, p.lng] as [number, number]),
   313→      ...shapes.flatMap((x) => x.vertices.map((v) => [v.lat, v.lng] as [number, number])),
   314→    ];

   318→

   336→
   337→  return (

   339→      <style>{`@media print { @page { size: A4 ${orientasi === "lanskap" ? "landscape" : "portrait"}; } }`}</style>

   341→        className="layout-sheet relative bg-white shadow-2xl border border-slate-300 print:border-0 print:shadow-none"
   342→        style={{ width: ukuran.w, height: ukuran.h, transformOrigin: "top center" }}

   366→        {/* Bingkai peta */}

   383→                  width: ukuranUtara,

   385→                onPointerDown={seretUtaraMulai}
   386→                onPointerMove={seretUtaraGerak}
   387→                onPointerUp={seretUtaraSelesai}
   388→                onPointerCancel={seretUtaraSelesai}

   392→                  <CompUtara className="block h-auto w-full" />
