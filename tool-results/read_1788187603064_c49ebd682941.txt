    15→    15→/** 1 px CSS = 0,264583 mm (96 DPI) — dasar perhitungan skala cetak A4. */
    16→    16→const MM_PER_PX = 25.4 / 96;
    17→    17→/** Meter per piksel pada zoom 0 Web Mercator (di ekuator). */
    18→    18→const MPP_Z0 = 156543.03392;
    19→    19→const ZOOM_MAKS = 25; // di atas ZOOM_TILE_ASLI citra di-upscale digital
    20→    20→const ZOOM_TILE_ASLI = 19;
    21→    21→
    22→    22→const derajatKeRadian = (d: number) => (d * Math.PI) / 180;
    23→    23→
    24→    24→/** Zoom pecahan agar 1 px kertas = MM_PER_PX × S mm di lapangan (skala 1:S). */
    25→    25→function zoomUntukSkala(s: number, lat: number, lebarPx: number): number {
    26→    26→  const mpp = (MM_PER_PX * s) / 1000; // meter lapangan per piksel kertas
    27→    27→  return Math.log2((MPP_Z0 * Math.cos(derajatKeRadian(lat))) / mpp);
    28→    28→}
    29→    29→
    30→    30→/** Skala 1:n aktual dari zoom saat ini pada latitude pusat peta. */
    31→    31→function skalaDariZoom(z: number, lat: number): number {
    32→    32→  const mpp = (MPP_Z0 * Math.cos(derajatKeRadian(lat))) / Math.pow(2, z);
    33→    33→  return Math.round((mpp * 1000) / MM_PER_PX);
    34→    34→}

    36→    36→/** Terima "1:50", "1/50", "50", "1 : 350" → angka pembagi skala. */
    37→    37→function parseSkala(teks: string): number | null {
    38→    38→  const b = teks.trim().replace(/,/g, ".").replace(/\s+/g, "");
    39→    39→  if (!b) return null;
    40→    40→  const m = b.match(/^1[:/x](\d+(?:\.\d+)?)$/i);
    41→    41→  if (m) return parseFloat(m[1]);
    42→    42→  const n = parseFloat(b);
    43→    43→  return Number.isFinite(n) && n >= 1 ? n : null;
    44→    44→}
    45→    45→

    51→    51→  return `${p(d.getDate())}/${p(d.getMonth() + 1)}/${d.getFullYear()}`;
    52→    52→};
