    35→/** Proyeksi lokal derajat ↔ meter di sekitar titik asal (cukup akurat untuk gambar lokal). */
    36→function buatProyeksi(o: LatLng) {
    37→  const mx = 111320 * Math.cos((o.lat * Math.PI) / 180);
    38→  const my = 110540;
    39→  return {
    40→    xy: (p: LatLng) => ({ x: (p.lng - o.lng) * mx, y: (p.lat - o.lat) * my }),
    41→    latlng: (x: number, y: number): LatLng => ({ lat: o.lat + y / my, lng: o.lng + x / mx }),
    42→  };
    43→}
    44→
    45→/** Vertiks lingkaran (poligon tertutup halus). */
    46→function titikLingkaran(c: LatLng, rMeter: number, n = N_LINGKARAN): LatLng[] {
    47→  const proy = buatProyeksi(c);
    48→  return Array.from({ length: n }, (_, i) => {
    49→    const a = (i / n) * 2 * Math.PI;
    50→    return proy.latlng(rMeter * Math.cos(a), rMeter * Math.sin(a));
    51→  });
    52→}
    53→
    54→/** Vertiks elips dari pusat + jari-jari horizontal & vertikal (meter). */
    55→function titikElips(c: LatLng, rx: number, ry: number, n = N_LINGKARAN): LatLng[] {
    56→  const proy = buatProyeksi(c);
    57→  return Array.from({ length: n }, (_, i) => {
    58→    const a = (i / n) * 2 * Math.PI;
    59→    return proy.latlng(rx * Math.cos(a), ry * Math.sin(a));
    60→  });
    61→}

    63→/**
    64→ * Busur SETENGAH lingkaran dari a ke b, menggelembung ke kiri/kanan arah jalan a→b
    65→ * (seperti belokan U di AutoCAD). Kiri = sisi kiri arah jalan.
    66→ */
    67→function titikBusurSetengah(a: LatLng, b: LatLng, arah: "kiri" | "kanan", n = N_BUSUR): LatLng[] {
    68→  const o: LatLng = { lat: (a.lat + b.lat) / 2, lng: (a.lng + b.lng) / 2 };
    69→  const proy = buatProyeksi(o);
    70→  const A = proy.xy(a);
    71→  const B = proy.xy(b);
    72→  const cx = (A.x + B.x) / 2;
    73→  const cy = (A.y + B.y) / 2;
    74→  const r = Math.hypot(B.x - A.x, B.y - A.y) / 2;
    75→  const t0 = Math.atan2(A.y - cy, A.x - cx);
    76→  // normal kiri arah jalan = sudut chord − 90° → geser sudut -1 (kiri) / +1 (kanan)
    77→  const sgn = arah === "kiri" ? -1 : 1;
    78→  const pts: LatLng[] = [];
    79→  for (let i = 0; i <= n; i++) {
    80→    const ang = t0 + sgn * (i / n) * Math.PI;
    81→    pts.push(proy.latlng(cx + r * Math.cos(ang), cy + r * Math.sin(ang)));
    82→  }
    83→  return pts;
    84→}

    86→/** Kurva Bezier kuadratik a→b lewat m (tanpa titik ujung — sudah ada di vertices). */
    87→function titikBezier2(a: LatLng, m: LatLng, b: LatLng, n = 24): LatLng[] {
    88→  const pts: LatLng[] = [];
    89→  for (let i = 1; i <= n; i++) {
    90→    const t = i / n;
    91→    const u = 1 - t;
    92→    pts.push({
    93→      lat: u * u * a.lat + 2 * u * t * m.lat + t * t * b.lat,
    94→      lng: u * u * a.lng + 2 * u * t * m.lng + t * t * b.lng,
    95→    });
    96→  }
    97→  return pts;
    98→}

   100→/** Jarak px titik ke ruas layar a-b (hit-test klik pada garis). */
   101→function jarakKeRuasPx(p: L.Point, a: L.Point, b: L.Point): number {
   102→  const dx = b.x - a.x;
   103→  const dy = b.y - a.y;
   104→  const len2 = dx * dx + dy * dy || 1;
   105→  let t = ((p.x - a.x) * dx + (p.y - a.y) * dy) / len2;
   106→  t = Math.max(0, Math.min(1, t));
   107→  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
   108→}

   110→/** Titik dalam poligon layar (ray casting) — hit-test klik pada poligon. */
   111→function dalamPoligonPx(p: L.Point, pts: L.Point[]): boolean {
   112→  let dalam = false;
   113→  for (let i = 0, j = pts.length - 1; i < pts.length; j = i++) {
   114→    const xi = pts[i].x;
   115→    const yi = pts[i].y;
   116→    const xj = pts[j].x;
   117→    const yj = pts[j].y;
   118→    const potong = yi > p.y !== yj > p.y && p.x < ((xj - xi) * (p.y - yi)) / (yj - yi) + xi;
   119→    if (potong) dalam = !dalam;
   120→  }
   121→  return dalam;
   122→}

   457→  // ---------- Alat edit bentuk: pindah titik + lengkungkan ruas lurus (ala AutoCAD) ----------
   458→  useEffect(() => {
   459→    if (tool !== "edit-bentuk") return;
   460→    const map = mapRef.current;
   461→    const l = layerRef.current;
   462→    if (!map || !l) return;
   463→
   464→    let shapeId: string | null = null;
   465→    let verts: LatLng[] = [];
   466→    let kind: "closed" | "open" = "open";
   467→    let bengkok: { seg: number; mid: LatLng } | null = null;
   468→    let handles: L.Marker[] = [];
   469→    let garis: L.Polyline | null = null;
   470→
   471→    const bersihkanHandle = () => {
   472→      handles.forEach((h) => l.temp.removeLayer(h));
   473→      handles = [];
   474→    };
   475→
   476→    /** Vertiks saat ini: ruas yang dibengkokkan diganti kurva Bezier. */
   477→    const verticesKini = (): LatLng[] => {
   478→      if (!bengkok) return verts;
   479→      const a = verts[bengkok.seg];
   480→      const b = verts[(bengkok.seg + 1) % verts.length];
   481→      if (!a || !b) return verts;
   482→      return [
   483→        ...verts.slice(0, bengkok.seg + 1),
   484→        ...titikBezier2(a, bengkok.mid, b),
   485→        ...verts.slice(bengkok.seg + 1),
   486→      ];
   487→    };
   488→
   489→    const gambarPratinjau = () => {
   490→      const vv = verticesKini().map((v) => [v.lat, v.lng] as [number, number]);
   491→      if (kind === "closed" && vv.length >= 3) vv.push(vv[0]);
   492→      if (!garis) {
   493→        garis = L.polyline(vv, { color: "#f59e0b", weight: 3, opacity: 0.95, interactive: false }).addTo(l.temp);
   494→      } else {

   496→      }
   497→    };
   498→
   499→    const bangunHandle = () => {
   500→      bersihkanHandle();
   501→      const min = kind === "closed" ? 3 : 2;
   502→      const n = verts.length;
   503→
   504→      // handle titik (kotak oranye): seret = pindah • Alt+klik = hapus

   515→          bangunHandle();
   516→          gambarPratinjau();
   517→        });
   518→        m.on("click", (e) => {
   519→          const ev = (e as unknown as L.LeafletMouseEvent).originalEvent;

   521→            verts.splice(i, 1);
   522→            useGis.getState().updateShape(shapeId!, { vertices: verts });
   523→            bangunHandle();
   524→            gambarPratinjau();
   525→          }
   526→        });
   527→        m.addTo(l.temp);
   528→        handles.push(m);
   529→      });
   530→
   531→      // handle tengah ruas (bulat biru): seret = lengkungkan ruas jadi busur
   532→      const jumlahRuas = kind === "closed" ? n : n - 1;
   533→      for (let i = 0; i < jumlahRuas; i++) {
   534→        const a = verts[i];
   535→        const b = verts[(i + 1) % n];
   536→        const mid: LatLng = { lat: (a.lat + b.lat) / 2, lng: (a.lng + b.lng) / 2 };
   537→        const m = L.marker([mid.lat, mid.lng], { icon: ikonHandle("tengah"), draggable: true, zIndexOffset: 900 });
   538→        m.on("dragstart", () => {
   539→          bengkok = { seg: i, mid: m.getLatLng() };
   540→          gambarPratinjau();
   541→        });
   542→        m.on("drag", () => {
   543→          if (bengkok) bengkok.mid = m.getLatLng();
   544→          gambarPratinjau();
   545→        });
   546→        m.on("dragend", () => {
   547→          if (bengkok) {
   548→            verts = verticesKini();
   549→            bengkok = null;
   550→            useGis.getState().updateShape(shapeId!, { vertices: verts });
   551→          }
   552→          bangunHandle();
   553→          gambarPratinjau();
   554→        });
   555→        m.addTo(l.temp);
   556→        handles.push(m);
   557→      }
   558→    };
   559→
   560→    const akhiriSesi = () => {
   561→      bersihkanHandle();
   562→      if (garis) {
   563→        l.temp.removeLayer(garis);
   564→        garis = null;
   565→      }
   566→      shapeId = null;
   567→      bengkok = null;
   568→    };

   570→    const mulai = (sh: GisShape) => {

   597→    window.addEventListener("geokita-edit-bentuk", onEditEv);
   598→
   599→    // klik peta: pilih bentuk yang kena (hit-test layar); klik area kosong = akhiri sesi
   600→    const onKlik = (e: L.LeafletMouseEvent) => {
   601→      const st = useGis.getState();
   602→      const cp = map.latLngToContainerPoint(e.latlng);
   603→      for (let si = st.shapes.length - 1; si >= 0; si--) {
   604→        const sh = st.shapes[si];
   605→        if (!sh.visible) continue;
   606→        const pts = sh.vertices.map((v) => map.latLngToContainerPoint([v.lat, v.lng]));
   607→        let minD = Infinity;
   608→        for (let i = 0; i < pts.length - 1; i++) minD = Math.min(minD, jarakKeRuasPx(cp, pts[i], pts[i + 1]));
   609→        if (sh.kind === "closed" && pts.length >= 3) {
   610→          if (dalamPoligonPx(cp, pts) || minD < 8) {
   611→            mulai(sh);
   612→            return;
   613→          }
   614→        } else if (minD < 8) {
   615→          mulai(sh);
   616→          return;
   617→        }
   618→      }
   619→      if (shapeId) akhiriSesi();
   620→    };
   621→
   622→    const onKey = (e: KeyboardEvent) => {
   623→      if (e.key === "Escape") useGis.getState().setTool(null);
   624→    };
   625→    map.on("click", onKlik);
   626→    window.addEventListener("keydown", onKey);
   627→
   628→    return () => {
   629→      map.off("click", onKlik);
   630→      window.removeEventListener("keydown", onKey);
   631→      window.removeEventListener("geokita-edit-bentuk", onEditEv);
   632→      akhiriSesi();
   633→    };
   634→  }, [tool]);

   636→  // ---------- Render titik ----------
   637→  useEffect(() => {

   639→    if (!l) return;

   642→    const cap = Math.min(points.length, RENDER_CAP);
   643→    for (let i = 0; i < cap; i++) {
   644→      const p = points[i];
   645→      const terpilih = selection.includes(p.id);
   646→      const marker = L.circleMarker([p.lat, p.lng], {
   647→        renderer,
   648→        radius: terpilih ? 7 : 5,
   649→        color: terpilih ? "#f59e0b" : "#1d4ed8",
   650→        weight: terpilih ? 2.5 : 1.5,
   651→        fillColor: terpilih ? "#fbbf24" : "#3b82f6",
   652→        fillOpacity: 0.9,
   653→      });
   654→      marker.on("click", () => {
   655→        const st = useGis.getState();
   656→        // mode blok: klik titik = pilih/hilangkan satu titik (tanpa popup)
   657→        if (st.tool === "select") {
   658→          klikFiturBarusan = true;
   659→          st.toggleSelect(p.id);

   662→        if (st.tool === "edit-bentuk") return; // alat edit bentuk: titik diabaikan

   665→      // Label nama sesuai mode: semua / terpilih (bertanda) / sembunyi
   666→      const labelNyala =
   667→        labelMode === "semua"
   668→          ? !!p.title

   670→            ? !!p.labelTampil && !!p.title

   673→        marker.bindTooltip(p.title, {
   674→          permanent: true,
   675→          direction: "right",
   676→          offset: [9, 0],

   693→      if (sh.kind === "closed" && latlngs.length >= 3) {
   694→        const poly = L.polygon(latlngs, {
   695→          color: terpilih ? "#f59e0b" : sh.color,

   735→          const st = useGis.getState();

   737→            klikFiturBarusan = true;

   742→            window.dispatchEvent(new CustomEvent("geokita-edit-bentuk", { detail: sh.id }));

   747→        const labelNyala =

   751→              ? !!sh.labelTampil && !!sh.title

   870→    if (semua.length === 0) return;
   871→    map.fitBounds(L.latLngBounds(semua).pad(0.15));

   873→
   874→  // Muat ulang ukuran peta saat kembali dari layout

   876→    if (view === "map") {

   878→    }
   879→  }, [view]);
   880→
   881→  return <div ref={containerRef} className="absolute inset-0 z-0" aria-label="Peta utama" role="application" />;
   882→}
